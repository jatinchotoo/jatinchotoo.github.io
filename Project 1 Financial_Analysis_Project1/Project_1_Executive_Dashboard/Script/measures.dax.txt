-- Revenue Growth (YoY)
Revenue Growth YoY =
VAR CurrentYearRevenue = SUM('Revenue'[Amount])
VAR PriorYearRevenue =
    CALCULATE(
        SUM('Revenue'[Amount]),
        SAMEPERIODLASTYEAR('Date'[Date])
    )
RETURN
DIVIDE(CurrentYearRevenue - PriorYearRevenue, PriorYearRevenue)

-- Return on Equity (ROE)
ROE =
DIVIDE(
    SUM('Financials'[NetIncome]),
    SUM('Financials'[Equity])
)

-- Debt-to-Equity Ratio
Debt to Equity =
DIVIDE(
    SUM('Financials'[TotalDebt]),
    SUM('Financials'[Equity])
)

-- Risk Flag (Traffic Light Logic)
Risk Flag =
SWITCH(
    TRUE(),
    [ROE] < 0.10, "🔴 High Risk",
    [ROE] < 0.15, "🟡 Moderate Risk",
    "🟢 Healthy"
)
